import * as React from "react";
import { cn } from "../../lib/utils";

interface NumberInputProps
  extends Omit<React.ComponentProps<"input">, "type" | "value" | "onChange"> {
  value: number;
  onValueChange: (next: number) => void;
}

export const NumberInput = React.forwardRef<HTMLInputElement, NumberInputProps>(
  ({ className, value, onValueChange, onBlur, onFocus, onKeyDown, ...props }, ref) => {
    const [draft, setDraft] = React.useState(String(value));
    const [editing, setEditing] = React.useState(false);

    React.useEffect(() => {
      if (!editing) {
        setDraft(String(value));
      }
    }, [value, editing]);

    const commit = React.useCallback(() => {
      setEditing(false);
      const trimmed = draft.trim();
      if (trimmed === "") {
        setDraft(String(value));
        return;
      }
      const parsed = Number(trimmed);
      if (Number.isFinite(parsed)) {
        onValueChange(parsed);
      } else {
        setDraft(String(value));
      }
    }, [draft, onValueChange, value]);

    return (
      <input
        ref={ref}
        type="number"
        value={draft}
        className={cn(
          "h-10 w-full rounded-xl border border-border bg-white px-3 text-sm text-foreground shadow-sm outline-none ring-primary placeholder:text-muted-foreground focus:ring-2",
          className
        )}
        onFocus={(event) => {
          setEditing(true);
          onFocus?.(event);
        }}
        onBlur={(event) => {
          commit();
          onBlur?.(event);
        }}
        onChange={(event) => {
          const raw = event.target.value;
          setDraft(raw);
          if (raw.trim() === "" || raw === "-" || raw === "." || raw === "-.") {
            return;
          }
          const parsed = Number(raw);
          if (Number.isFinite(parsed)) {
            onValueChange(parsed);
          }
        }}
        onKeyDown={(event) => {
          if (event.key === "Enter") {
            commit();
          } else if (event.key === "Escape") {
            setDraft(String(value));
            setEditing(false);
          }
          onKeyDown?.(event);
        }}
        {...props}
      />
    );
  }
);

NumberInput.displayName = "NumberInput";

function isoWeekFromUtcDate(date: Date): number {
  const target = new Date(
    Date.UTC(date.getUTCFullYear(), date.getUTCMonth(), date.getUTCDate())
  );
  const dayNumber = target.getUTCDay() || 7;
  target.setUTCDate(target.getUTCDate() + 4 - dayNumber);
  const yearStart = new Date(Date.UTC(target.getUTCFullYear(), 0, 1));
  return Math.ceil(((target.getTime() - yearStart.getTime()) / 86400000 + 1) / 7);
}

function mulberry32(seed: number): () => number {
  let t = seed >>> 0;
  return () => {
    t += 0x6d2b79f5;
    let n = Math.imul(t ^ (t >>> 15), 1 | t);
    n ^= n + Math.imul(n ^ (n >>> 7), 61 | n);
    return ((n ^ (n >>> 14)) >>> 0) / 4294967296;
  };
}

function randomInt(rng: () => number, min: number, max: number): number {
  return Math.floor(rng() * (max - min + 1)) + min;
}

function pickUnique<T>(items: T[], count: number, rng: () => number): T[] {
  const copy = [...items];
  for (let i = copy.length - 1; i > 0; i -= 1) {
    const j = Math.floor(rng() * (i + 1));
    [copy[i], copy[j]] = [copy[j], copy[i]];
  }
  return copy.slice(0, Math.min(count, copy.length));
}

function isoWeekFromUtcDate(date: Date): number {
  const target = new Date(
    Date.UTC(date.getUTCFullYear(), date.getUTCMonth(), date.getUTCDate())
  );
  const dayNumber = target.getUTCDay() || 7;
  target.setUTCDate(target.getUTCDate() + 4 - dayNumber);
  const yearStart = new Date(Date.UTC(target.getUTCFullYear(), 0, 1));
  return Math.ceil(((target.getTime() - yearStart.getTime()) / 86400000 + 1) / 7);
}

export function generateDemoProject(seed = 1337, year = 2026): ProjectData {
  const rng = mulberry32(seed);
  const now = nowIso();
  let idCounter = 1;
  const nextId = (prefix: string): string => `${prefix}-${seed}-${idCounter++}`;

  const groupSpecs = [
    { name: "Team A", workerCount: 50 },
    { name: "Team B", workerCount: 50 },
    { name: "Team C", workerCount: 50 },
    { name: "Team D", workerCount: 100 }
  ];
  const groups = groupSpecs.map((spec) => ({
    id: nextId("grp"),
    name: spec.name,
    minHeadcountPerWeek: Math.max(
      1,
      spec.workerCount - randomInt(rng, Math.ceil(spec.workerCount * 0.15), Math.ceil(spec.workerCount * 0.28))
    )
  }));

  const skills: Skill[] = Array.from({ length: 16 }, (_, idx) => ({
    id: nextId("skill"),
    name: `Skill ${idx + 1}`
  }));

  let workerIndex = 1;
  const workers: Worker[] = groups.flatMap((group, index) => {
    const count = groupSpecs[index].workerCount;
    return Array.from({ length: count }, () => {
      const skillCount = randomInt(rng, 1, 6);
      const worker: Worker = {
        id: nextId("wrk"),
        name: `Worker ${workerIndex}`,
        groupId: group.id,
        skillIds: pickUnique(skills.map((skill) => skill.id), skillCount, rng).sort()
      };
      workerIndex += 1;
      return worker;
    });
  });

  const skillRequirements = groups.flatMap((group) => {
    const requirementCount = randomInt(rng, 2, 4);
    const pickedSkills = pickUnique(skills, requirementCount, rng);
    const groupWorkerCount = workers.filter((worker) => worker.groupId === group.id).length;
    return pickedSkills.map((skill) => ({
      id: nextId("req"),
      groupId: group.id,
      skillId: skill.id,
      minPerWeek: randomInt(rng, Math.max(2, Math.floor(groupWorkerCount * 0.1)), Math.max(3, Math.floor(groupWorkerCount * 0.2)))
    }));
  });

  const aprilWeek = isoWeekFromUtcDate(new Date(Date.UTC(year, 3, 1)));
  const septemberWeek = isoWeekFromUtcDate(new Date(Date.UTC(year, 8, 30)));
  const vacationLength = 4;
  const requestStartMin = Math.max(1, Math.min(53 - vacationLength + 1, aprilWeek));
  const requestStartMax = Math.max(
    requestStartMin,
    Math.min(53 - vacationLength + 1, septemberWeek - vacationLength + 1)
  );

  const requests = workers.map((worker) => {
    const startWeek = randomInt(rng, requestStartMin, requestStartMax);
    return {
      id: nextId("reqv"),
      workerId: worker.id,
      startWeek,
      endWeek: startWeek + vacationLength - 1,
      priority: 2 as const
    };
  });

  const baseProject: ProjectData = {
    schemaVersion: 1,
    project: {
      id: `demo-${seed}`,
      name: "Demo 250 workers",
      createdAt: now,
      updatedAt: now
    },
    timeWindow: {
      year,
      startWeek: requestStartMin,
      endWeek: septemberWeek,
      includeMonths: [4, 5, 6, 7, 8, 9]
    },
    groups,
    skills,
    skillRequirements,
    workers,
    requests,
    plan: {
      assignments: {},
      groupOverrides: {},
      generatedAt: now
    },
    violations: [],
    unmetRequests: []
  };

  const generated = generatePlan(baseProject);
  return {
    ...baseProject,
    plan: generated.plan,
    violations: generated.violations,
    unmetRequests: generated.unmetRequests
  };
}

export function importProjectFromJson(rawJson: string): ProjectData {
  let parsed: unknown;
  try {
    parsed = JSON.parse(rawJson);
  } catch {
    throw new Error("Import failed: tiedosto ei ollut validia JSONia.");
  }

  const migrated = migrateToV1(parsed);
  const validation = validateProjectV1(migrated);
  if (!validation.valid || !validation.value) {
    throw new Error(`Import failed: ${validation.errors.join(" ")}`);
  }
  return validation.value;
}

function toRanges(weeks: number[]): string {
  if (weeks.length === 0) {
    return "none";
  }
  const sorted = [...new Set(weeks)].sort((a, b) => a - b);
  const ranges: string[] = [];
  let start = sorted[0];
  let previous = sorted[0];
  for (let i = 1; i < sorted.length; i += 1) {
    const week = sorted[i];
    if (week === previous + 1) {
      previous = week;
      continue;
    }
    ranges.push(start === previous ? `W${start}` : `W${start}-W${previous}`);
    start = week;
    previous = week;
  }
  ranges.push(start === previous ? `W${start}` : `W${start}-W${previous}`);
  return ranges.join(", ");
}

function wrapText(text: string, font: PDFFont, size: number, maxWidth: number): string[] {
  if (!text) {
    return [""];
  }
  const words = text.split(/\s+/);
  const lines: string[] = [];
  let current = "";
  for (const word of words) {
    const next = current ? `${current} ${word}` : word;
    if (font.widthOfTextAtSize(next, size) <= maxWidth) {
      current = next;
      continue;
    }
    if (current) {
      lines.push(current);
      current = word;
      continue;
    }
    let part = "";
    for (const char of word) {
      const candidate = `${part}${char}`;
      if (font.widthOfTextAtSize(candidate, size) <= maxWidth) {
        part = candidate;
      } else {
        lines.push(part);
        part = char;
      }
    }
    current = part;
  }
  if (current) {
    lines.push(current);
  }
  return lines;
}

export async function exportProjectToPdf(
  project: ProjectData,
  options?: { mode?: PdfExportMode }
): Promise<{
  filename: string;
  bytes: Uint8Array;
}> {
  const mode: PdfExportMode = options?.mode ?? "full";
  const pdf = await PDFDocument.create();
  const fontRegular = await pdf.embedFont(StandardFonts.Helvetica);
  const fontBold = await pdf.embedFont(StandardFonts.HelveticaBold);

  const pageWidth = 595.28;
  const pageHeight = 841.89;
  const margin = 40;
  const contentWidth = pageWidth - margin * 2;
  let page: PDFPage = pdf.addPage([pageWidth, pageHeight]);
  let y = pageHeight - margin;

  const ensureSpace = (heightNeeded: number) => {
    if (y - heightNeeded >= margin) {
      return;
    }
    page = pdf.addPage([pageWidth, pageHeight]);
    y = pageHeight - margin;
  };

  const writeLine = (
    text: string,
    options: {
      bold?: boolean;
      size?: number;
      color?: { r: number; g: number; b: number };
      indent?: number;
      gapAfter?: number;
    } = {}
  ) => {
    const size = options.size ?? 10;
    const lineHeight = size + 4;
    const font = options.bold ? fontBold : fontRegular;
    const indent = options.indent ?? 0;
    const maxWidth = contentWidth - indent;
    const lines = wrapText(text, font, size, maxWidth);
    for (const line of lines) {
      ensureSpace(lineHeight);
      page.drawText(line, {
        x: margin + indent,
        y,
        size,
        font,
        color: options.color ? rgb(options.color.r, options.color.g, options.color.b) : rgb(0, 0, 0)
      });
      y -= lineHeight;
    }
    if (options.gapAfter) {
      y -= options.gapAfter;
    }
  };

  const weeks = listWeeks(project.timeWindow);
  const groupMap = new Map(project.groups.map((group) => [group.id, group.name]));
  const skillMap = new Map(project.skills.map((skill) => [skill.id, skill.name]));

  writeLine("Lomalistat - Project PDF Export", {
    bold: true,
    size: 16,
    color: { r: 0.06, g: 0.19, b: 0.39 },
    gapAfter: 4
  });
  writeLine(mode === "vacations_only" ? "Mode: Vacations only" : "Mode: Full report");
  writeLine(`Project: ${project.project.name}`, { bold: true });
  writeLine(
    `Time window: ${project.timeWindow.year}, weeks W${project.timeWindow.startWeek}-W${project.timeWindow.endWeek}`
  );
  writeLine(`Groups: ${project.groups.length}, Workers: ${project.workers.length}`);
  writeLine(`Requests: ${project.requests.length}, Violations: ${project.violations.length}`);
  writeLine(`Unmet requests: ${project.unmetRequests.length}`, { gapAfter: 6 });

  if (mode === "full") {
    writeLine("Groups", { bold: true, size: 12 });
    for (const group of project.groups) {
      writeLine(
        `- ${group.name}: min headcount per week ${group.minHeadcountPerWeek}`
      );
    }
    writeLine("", { gapAfter: 2 });

    writeLine("Skill requirements", { bold: true, size: 12 });
    if (project.skillRequirements.length === 0) {
      writeLine("- none");
    } else {
      const sortedRequirements = [...project.skillRequirements].sort((a, b) => {
        if (a.groupId !== b.groupId) {
          return a.groupId.localeCompare(b.groupId);
        }
        return a.skillId.localeCompare(b.skillId);
      });
      for (const requirement of sortedRequirements) {
        writeLine(
          `- ${groupMap.get(requirement.groupId) ?? requirement.groupId}: ${
            skillMap.get(requirement.skillId) ?? requirement.skillId
          } min ${requirement.minPerWeek}/week`
        );
      }
    }
    writeLine("", { gapAfter: 2 });
  }

  writeLine("Workers and planned vacation weeks", { bold: true, size: 12 });
  const sortedWorkers = [...project.workers].sort((a, b) => a.name.localeCompare(b.name));
  for (const worker of sortedWorkers) {
    const workerAssignments = project.plan.assignments[worker.id] ?? {};

      );
    }
  }
  writeLine("", { gapAfter: 2 });

    writeLine("Top violations", { bold: true, size: 12 });
    if (project.violations.length === 0) {
      writeLine("- none");
    } else {
      const sortedViolations = [...project.violations]

        .slice(0, 200);

          `- W${violation.week} ${groupMap.get(violation.groupId) ?? violation.groupId} ${
            violation.type
          }${violation.skillId ? ` ${skillMap.get(violation.skillId) ?? violation.skillId}` : ""}: req ${
            violation.required
          }, actual ${violation.actual}, missing ${violation.missing}`
        );
      }
    }
  }

  return {
    filename: createPdfFilename(project, mode),
    bytes
  };

export function buildDefaultAssignments(project: ProjectData): ProjectData["plan"]["assignments"] {
  const assignments: ProjectData["plan"]["assignments"] = {};
  const weeks = listWeeks(project.timeWindow);

  for (const worker of project.workers) {
    assignments[worker.id] = {};
    for (const week of weeks) {
      assignments[worker.id][week] = "WORK";
    }
  }

  return assignments;
}

export function getAssignment(
  assignments: ProjectData["plan"]["assignments"],
  workerId: string,
  week: number
): AssignmentState {
  return assignments[workerId]?.[week] ?? "WORK";
}

function workerHasGroup(project: ProjectData, groupId: string): boolean {
  return project.groups.some((group) => group.id === groupId);
}

export function getWorkerEffectiveGroupForWeek(
  project: ProjectData,
  assignments: ProjectData["plan"]["assignments"],
  groupOverrides: GroupOverrides,
  workerId: string,
  week: number
): string | null {
  const worker = project.workers.find((item) => item.id === workerId);
  if (!worker) {
    return null;
  }
  if (getAssignment(assignments, workerId, week) !== "WORK") {
    return null;
  }

  const overrideGroupId = groupOverrides[workerId]?.[week];
  if (overrideGroupId && workerHasGroup(project, overrideGroupId)) {
    return overrideGroupId;
  }

  return worker.groupId;
}

): number {

      project,
      assignments,
      groupOverrides,
      worker.id,

function requirementsForGroup(

  groupId: string

export function evaluateGroupWeek(
  project: ProjectData,
  assignments: ProjectData["plan"]["assignments"],
  groupId: string,
  week: number,
  groupOverrides: GroupOverrides = project.plan.groupOverrides ?? {}
): { ok: true } | { ok: false; failure: ConstraintFailure } {
  const group = project.groups.find((item) => item.id === groupId);
  if (!group) {
    return { ok: true };
  }

  const headcountActual = countWorkersWorkingInGroup(project, assignments, groupId, week, groupOverrides);
  if (headcountActual < group.minHeadcountPerWeek) {
    return {
      ok: false,
      failure: {
        type: "HEADCOUNT",
        groupId,
        required: group.minHeadcountPerWeek,
        actual: headcountActual,
        missing: group.minHeadcountPerWeek - headcountActual
      }
    };
  }

  for (const requirement of requirementsForGroup(project, groupId)) {
    const actual = countWorkersWithSkillWorking(
      project,
      assignments,
      groupId,
      week,
      requirement.skillId,
      groupOverrides
    );
    if (actual < requirement.minPerWeek) {
      return {
        ok: false,
        failure: {
          type: "SKILL",
          groupId,
          skillId: requirement.skillId,
          required: requirement.minPerWeek,
          actual,
          missing: requirement.minPerWeek - actual
        }
      };
    }
  }

  return { ok: true };
}

export function computeViolations(
  project: ProjectData,
  assignments: ProjectData["plan"]["assignments"] = project.plan.assignments,
  groupOverrides: GroupOverrides = project.plan.groupOverrides ?? {}
): Violation[] {
  const violations: Violation[] = [];
  const weeks = listWeeks(project.timeWindow);

  for (const group of project.groups) {
    const reqs = requirementsForGroup(project, group.id);
    for (const week of weeks) {
      const actualHeadcount = countWorkersWorkingInGroup(
        project,
        assignments,
        group.id,
        week,
        groupOverrides
      );
      if (actualHeadcount < group.minHeadcountPerWeek) {
        violations.push({
          id: `v-${group.id}-${week}-HEADCOUNT`,
          week,
          groupId: group.id,
          type: "HEADCOUNT",
          required: group.minHeadcountPerWeek,
          actual: actualHeadcount,
          missing: group.minHeadcountPerWeek - actualHeadcount
        });
      }

      for (const req of reqs) {
        const actualSkill = countWorkersWithSkillWorking(
          project,
          assignments,
          group.id,
          week,
          req.skillId,
          groupOverrides
        );
        if (actualSkill < req.minPerWeek) {
          violations.push({
            id: `v-${group.id}-${week}-SKILL-${req.skillId}`,
            week,
            groupId: group.id,
            type: "SKILL",
            skillId: req.skillId,
            required: req.minPerWeek,
            actual: actualSkill,
            missing: req.minPerWeek - actualSkill
          });
        }
      }
    }
  }

  return violations;
}

export function computeCoverage(
  project: ProjectData,
  assignments: ProjectData["plan"]["assignments"] = project.plan.assignments,
  groupOverrides: GroupOverrides = project.plan.groupOverrides ?? {}
): GroupCoverageRow[] {
  const rows: GroupCoverageRow[] = [];
  const skillNameById = new Map(project.skills.map((skill) => [skill.id, skill.name]));
  const weeks = listWeeks(project.timeWindow);

  for (const group of project.groups) {
    const reqs = requirementsForGroup(project, group.id);
    for (const week of weeks) {
      const headcountActual = countWorkersWorkingInGroup(
        project,
        assignments,
        group.id,
        week,
        groupOverrides
      );
      rows.push({
        week,
        groupId: group.id,
        groupName: group.name,
        headcount: {
          required: group.minHeadcountPerWeek,
          actual: headcountActual,
          missing: Math.max(0, group.minHeadcountPerWeek - headcountActual)
        },
        skills: reqs.map((req) => {
          const actual = countWorkersWithSkillWorking(
            project,
            assignments,
            group.id,
            week,
            req.skillId,
            groupOverrides
          );
          return {
            skillId: req.skillId,
            skillName: skillNameById.get(req.skillId) ?? req.skillId,
            required: req.minPerWeek,
            actual,
            missing: Math.max(0, req.minPerWeek - actual)
          };
        })
      });
    }
  }

  return rows;
}

import type { ProjectData, UnmetRequest } from "../model/types";
import { nowIso } from "../model/id";
import { isWeekInTimeWindow } from "../model/weeks";

export interface GeneratePlanResult {
  plan: ProjectData["plan"];
  violations: ProjectData["violations"];
  unmetRequests: UnmetRequest[];

interface GeneratePlanOptions {

function replaceOverrides(
  target: ProjectData["plan"]["groupOverrides"],
  source: ProjectData["plan"]["groupOverrides"]
): void {
  for (const key of Object.keys(target)) {
    delete target[key];
  }
  Object.assign(target, source);
}

function setWorkerOverrideForWeek(
  overrides: ProjectData["plan"]["groupOverrides"],
  workerId: string,
  workerHomeGroupId: string,
  week: number,
  targetGroupId: string | null
): void {
  if (!targetGroupId || targetGroupId === workerHomeGroupId) {
    if (overrides[workerId]) {
      delete overrides[workerId][week];
      if (Object.keys(overrides[workerId]).length === 0) {
        delete overrides[workerId];
      }
    }
    return;
  }

  overrides[workerId] ??= {};
  overrides[workerId][week] = targetGroupId;
}

function missingForFailureType(
  project: ProjectData,
  assignments: ProjectData["plan"]["assignments"],
  overrides: ProjectData["plan"]["groupOverrides"],
  groupId: string,
  week: number,
  failure:
    | {
        type: "HEADCOUNT";
        required: number;
      }
    | {
        type: "SKILL";
        required: number;
        skillId: string;
      }
): number {
  if (failure.type === "HEADCOUNT") {
    const actual = countWorkersWorkingInGroup(project, assignments, groupId, week, overrides);
    return Math.max(0, failure.required - actual);
  }

  const actual = countWorkersWithSkillWorking(
    project,
    assignments,
    groupId,
    week,
    failure.skillId,
    overrides
  );
  return Math.max(0, failure.required - actual);
}

function pickBestBackfillCandidate(
  project: ProjectData,
  assignments: ProjectData["plan"]["assignments"],
  overrides: ProjectData["plan"]["groupOverrides"],
  targetGroupId: string,
  week: number,
  failure: {
    type: "HEADCOUNT" | "SKILL";
    required: number;
    missing: number;
    skillId?: string;
  },
  excludedWorkerIds: Set<string>
): { workerId: string } | null {
  const sortedWorkers = [...project.workers].sort((a, b) => a.id.localeCompare(b.id));
  let best: { workerId: string; remainingMissing: number } | null = null;

  for (const worker of sortedWorkers) {
    if (excludedWorkerIds.has(worker.id)) {
      continue;
    }
    if ((assignments[worker.id]?.[week] ?? "WORK") !== "WORK") {
      continue;
    }

    const donorGroupId = getWorkerEffectiveGroupForWeek(
      project,
      assignments,
      overrides,
      worker.id,
      week
    );
    if (!donorGroupId || donorGroupId === targetGroupId) {
      continue;
    }

    if (failure.type === "SKILL" && failure.skillId && !worker.skillIds.includes(failure.skillId)) {
      continue;
    }

    const trialOverrides = structuredClone(overrides);
    setWorkerOverrideForWeek(trialOverrides, worker.id, worker.groupId, week, targetGroupId);

    const donorOk = evaluateGroupWeek(project, assignments, donorGroupId, week, trialOverrides);
    if (!donorOk.ok) {
      continue;
    }

    const remainingMissing =
      failure.type === "HEADCOUNT"
        ? missingForFailureType(project, assignments, trialOverrides, targetGroupId, week, {
            type: "HEADCOUNT",
            required: failure.required
          })
        : missingForFailureType(project, assignments, trialOverrides, targetGroupId, week, {
            type: "SKILL",
            required: failure.required,
            skillId: failure.skillId ?? ""
          });

    if (remainingMissing >= failure.missing) {
      continue;
    }

    if (!best || remainingMissing < best.remainingMissing) {
      best = {
        workerId: worker.id,
        remainingMissing
      };
    }
  }

  return best;
}

function tryAutoBackfillForWeek(
  project: ProjectData,
  assignments: ProjectData["plan"]["assignments"],
  overrides: ProjectData["plan"]["groupOverrides"],
  targetGroupId: string,
  week: number,
  excludedWorkerIds: Set<string>
): { ok: true } | { ok: false; failure: ConstraintFailure } {
  const excluded = new Set(excludedWorkerIds);

  for (let i = 0; i < project.workers.length; i += 1) {
    const validation = evaluateGroupWeek(project, assignments, targetGroupId, week, overrides);
    if (validation.ok) {
      return { ok: true };
    }

    const candidate = pickBestBackfillCandidate(
      project,
      assignments,
      overrides,
      targetGroupId,
      week,
      validation.failure,
      excluded
    );
    if (!candidate) {
      return { ok: false, failure: validation.failure };
    }

    const worker = project.workers.find((item) => item.id === candidate.workerId);
    if (!worker) {
      return { ok: false, failure: validation.failure };
    }
    setWorkerOverrideForWeek(overrides, worker.id, worker.groupId, week, targetGroupId);
    excluded.add(worker.id);
  }

  const finalValidation = evaluateGroupWeek(project, assignments, targetGroupId, week, overrides);
  if (finalValidation.ok) {
    return { ok: true };
  }

  return { ok: false, failure: finalValidation.failure };
}

function runGeneratePlan(project: ProjectData, options: GeneratePlanOptions): GeneratePlanResult {
  const assignments = buildDefaultAssignments(project);
  const groupOverrides: ProjectData["plan"]["groupOverrides"] = {};
  const sortedRequests = [...project.requests].sort((a, b) => {
    if (a.priority !== b.priority) {
      return a.priority - b.priority;
    }
    return a.id.localeCompare(b.id);
  });
  const unmetRequests: UnmetRequest[] = [];

  for (const request of sortedRequests) {
    const worker = project.workers.find((item) => item.id === request.workerId);
    if (!worker) {
      continue;
    }

    for (let week = request.startWeek; week <= request.endWeek; week += 1) {
      if (!isWeekInTimeWindow(week, project.timeWindow)) {
        continue;
      }
      if ((assignments[worker.id]?.[week] ?? "WORK") === "VACATION") {
        continue;
      }

      const evaluatedGroupId =
        getWorkerEffectiveGroupForWeek(project, assignments, groupOverrides, worker.id, week) ??
        worker.groupId;
      const previousAssignment = assignments[worker.id]?.[week] ?? "WORK";
      const snapshotOverrides = structuredClone(groupOverrides);
      assignments[worker.id] ??= {};
      assignments[worker.id][week] = "VACATION";
      setWorkerOverrideForWeek(groupOverrides, worker.id, worker.groupId, week, null);

      let validation = evaluateGroupWeek(project, assignments, evaluatedGroupId, week, groupOverrides);

      if (!validation.ok && options.allowTransfers) {
        const transferResult = tryAutoBackfillForWeek(
          project,
          assignments,
          groupOverrides,
          evaluatedGroupId,
          week,
          new Set([worker.id])
        );
        if (transferResult.ok) {
          continue;
        }
        validation = {
          ok: false,
          failure: transferResult.failure
        };
      }

      if (validation.ok) {
        continue;
      }

      assignments[worker.id][week] = previousAssignment;
      replaceOverrides(groupOverrides, snapshotOverrides);
      unmetRequests.push({
        id: `ur-${request.id}-${week}`,
        requestId: request.id,
        workerId: request.workerId,
        week,
        reasonType: validation.failure.type,
        groupId: validation.failure.groupId,
        skillId: validation.failure.skillId,
        required: validation.failure.required,
        actual: validation.failure.actual,
        missing: validation.failure.missing
      });
    }
  }

  const plan: ProjectData["plan"] = {
    assignments,
    groupOverrides,
    generatedAt: nowIso()
  };

  const violations = computeViolations(project, assignments, groupOverrides);
  return {
    plan,
    violations,
    unmetRequests
  };
}

export function generatePlan(project: ProjectData): GeneratePlanResult {
  return runGeneratePlan(project, { allowTransfers: false });
}

export function generatePlanWithTransfers(project: ProjectData): GeneratePlanResult {
  return runGeneratePlan(project, { allowTransfers: true });
}

function recomputeUnmetRequests(
  project: ProjectData,
  assignments: ProjectData["plan"]["assignments"],
  groupOverrides: ProjectData["plan"]["groupOverrides"]
): UnmetRequest[] {
  const unmetRequests: UnmetRequest[] = [];
  const sortedRequests = [...project.requests].sort((a, b) => {
    if (a.priority !== b.priority) {
      return a.priority - b.priority;
    }
    return a.id.localeCompare(b.id);
  });

  for (const request of sortedRequests) {
    const worker = project.workers.find((item) => item.id === request.workerId);
    if (!worker) {
      continue;
    }
    const workerAssignments = assignments[worker.id] ?? {};
    assignments[worker.id] = workerAssignments;

    for (let week = request.startWeek; week <= request.endWeek; week += 1) {
      if (!isWeekInTimeWindow(week, project.timeWindow)) {
        continue;
      }
      if (workerAssignments[week] === "VACATION") {
        continue;
      }

      const previous = workerAssignments[week] ?? "WORK";
      workerAssignments[week] = "VACATION";
      const evaluatedGroupId =
        getWorkerEffectiveGroupForWeek(project, assignments, groupOverrides, worker.id, week) ??
        worker.groupId;
      const result = evaluateGroupWeek(project, assignments, evaluatedGroupId, week, groupOverrides);
      workerAssignments[week] = previous;
      if (result.ok) {
        continue;
      }

      unmetRequests.push({
        id: `ur-${request.id}-${week}`,
        requestId: request.id,
        workerId: request.workerId,
        week,
        reasonType: result.failure.type,
        groupId: result.failure.groupId,
        skillId: result.failure.skillId,
        required: result.failure.required,
        actual: result.failure.actual,
        missing: result.failure.missing
      });
    }
  }

  return unmetRequests;
}

export function debounce<T extends (...args: Parameters<T>) => void>(fn: T, waitMs: number) {
  let timer: ReturnType<typeof setTimeout> | undefined;

function isAbortError(error: unknown): boolean {
  return error instanceof DOMException && error.name === "AbortError";
}

function pickerTypesFromAccept(acceptType?: PickerAcceptType) {
  if (!acceptType) {
    return undefined;
  }
  return [
    {
      description: acceptType.description,
      accept: {
        [acceptType.mimeType]: acceptType.extensions
      }
    }
  ];
}

export function isSavePickerSupported(): boolean {
  return Boolean((window as SavePickerWindow).showSaveFilePicker);
}

export async function pickSaveFileHandle(
  filename: string,
  acceptType?: PickerAcceptType
): Promise<SaveFileHandle | null> {
  const picker = (window as SavePickerWindow).showSaveFilePicker;
  if (!picker) {
    return null;
  }

  try {
    const handle = await picker({
      suggestedName: filename,
      types: pickerTypesFromAccept(acceptType)
    });
    return handle;
  } catch (error) {
    if (isAbortError(error)) {
      return null;
    }
    throw error;
  }
}

export async function writeBlobToFileHandle(handle: SaveFileHandle, blob: Blob): Promise<void> {
  const writable = await handle.createWritable();
  await writable.write(blob);
  await writable.close();
}

function asRecord(input: unknown): Record<string, unknown> | null {
  if (typeof input !== "object" || input === null || Array.isArray(input)) {
    return null;
  }
  return input as Record<string, unknown>;
}

function asString(value: unknown, fallback: string): string {
  if (typeof value === "string" && value.trim()) {
    return value;
  }
  return fallback;
}

function asNumber(value: unknown, fallback: number): number {
  if (typeof value === "number" && Number.isFinite(value)) {
    return value;
  }
  return fallback;
}

export function migrateToV1(input: unknown): ProjectData {
  const now = nowIso();
  const source = asRecord(input);
  if (!source) {
    throw new Error("Imported JSON is not an object.");
  }

  if (source.schemaVersion !== undefined && source.schemaVersion !== 1) {
    throw new Error(`Unsupported schemaVersion: ${String(source.schemaVersion)}`);
  }

  const projectSource = asRecord(source.project) ?? {};
  const timeWindowSource = asRecord(source.timeWindow) ?? {};
  const planSource = asRecord(source.plan) ?? {};

  return {
    schemaVersion: 1,
    project: {
      id: asString(projectSource.id, `import-${Date.now().toString(36)}`),
      name: asString(projectSource.name, "Tuotu projekti"),
      createdAt: asString(projectSource.createdAt, now),
      updatedAt: asString(projectSource.updatedAt, now)
    },
    timeWindow: normalizeTimeWindow({
      year: asNumber(timeWindowSource.year, new Date().getFullYear()),
      startWeek: asNumber(timeWindowSource.startWeek, 1),
      endWeek: asNumber(timeWindowSource.endWeek, 52),
      includeMonths: Array.isArray(timeWindowSource.includeMonths)
        ? timeWindowSource.includeMonths
            .map((item) => asNumber(item, 0))
            .filter((month) => month >= 1 && month <= 12)
        : undefined
    }),
    groups: Array.isArray(source.groups) ? (source.groups as ProjectData["groups"]) : [],
    skills: Array.isArray(source.skills) ? (source.skills as ProjectData["skills"]) : [],
    skillRequirements: Array.isArray(source.skillRequirements)
      ? (source.skillRequirements as ProjectData["skillRequirements"])
      : [],
    workers: Array.isArray(source.workers) ? (source.workers as ProjectData["workers"]) : [],
    requests: Array.isArray(source.requests) ? (source.requests as ProjectData["requests"]) : [],
    plan: {
      assignments:
        typeof planSource.assignments === "object" && planSource.assignments !== null
          ? (planSource.assignments as ProjectData["plan"]["assignments"])
          : {},
      groupOverrides:
        typeof planSource.groupOverrides === "object" && planSource.groupOverrides !== null
          ? (planSource.groupOverrides as ProjectData["plan"]["groupOverrides"])
          : {},
      generatedAt: asString(planSource.generatedAt, now),
      notes: typeof planSource.notes === "string" ? planSource.notes : undefined
    },
    violations: Array.isArray(source.violations)

  if (errors.length > 0) {

import type { TimeWindow } from "./types";

export function clampWeek(week: number): number {
  if (!Number.isFinite(week)) {
    return 1;
  }
  return Math.min(53, Math.max(1, Math.trunc(week)));
}

export function normalizeTimeWindow(timeWindow: TimeWindow): TimeWindow {
  const startWeek = clampWeek(timeWindow.startWeek);
  const endWeek = clampWeek(timeWindow.endWeek);
  return {
    ...timeWindow,
    startWeek: Math.min(startWeek, endWeek),
    endWeek: Math.max(startWeek, endWeek)
  };
}

export function listWeeks(timeWindow: TimeWindow): number[] {
  const normalized = normalizeTimeWindow(timeWindow);
  const weeks: number[] = [];
  for (let week = normalized.startWeek; week <= normalized.endWeek; week += 1) {
    weeks.push(week);
  }
  return weeks;
}

export function weekCount(timeWindow: TimeWindow): number {
  return listWeeks(timeWindow).length;
}

export function isWeekInTimeWindow(week: number, timeWindow: TimeWindow): boolean {
  const normalized = normalizeTimeWindow(timeWindow);
  return week >= normalized.startWeek && week <= normalized.endWeek;
}

interface BackfillSuggestion {
  workerId: string;
  workerName: string;
  fromGroupId: string;
  fromGroupName: string;
  targetActualAfter: number;
  remainingMissing: number;
  donorSafe: boolean;
}

function statusVariant(missing: number): "success" | "warning" | "danger" {
  if (missing <= 0) {
    return "success";
  }
  if (missing === 1) {
    return "warning";
  }
  return "danger";
}

function setOverrideForWeek(
  base: ProjectData["plan"]["groupOverrides"],
  workerId: string,
  week: number,
  workerHomeGroupId: string,
  targetGroupId: string
): ProjectData["plan"]["groupOverrides"] {
  const next: ProjectData["plan"]["groupOverrides"] = {
    ...base,
    [workerId]: {
      ...(base[workerId] ?? {})
    }
  };

  if (targetGroupId === workerHomeGroupId) {
    delete next[workerId][week];
    if (Object.keys(next[workerId]).length === 0) {
      delete next[workerId];
    }
  } else {
    next[workerId][week] = targetGroupId;
  }

  return next;
}

function buildBackfillSuggestions(
  project: ProjectData,
  violation: Violation
): BackfillSuggestion[] {
  if (violation.missing <= 0) {
    return [];
  }

  const baseOverrides = project.plan.groupOverrides ?? {};
  const assignments = project.plan.assignments;
  const groupNameById = new Map(project.groups.map((group) => [group.id, group.name]));
  const suggestions: BackfillSuggestion[] = [];

  for (const worker of project.workers) {
    if ((assignments[worker.id]?.[violation.week] ?? "WORK") !== "WORK") {
      continue;
    }

    const currentGroupId = getWorkerEffectiveGroupForWeek(
      project,
      assignments,
      baseOverrides,
      worker.id,
      violation.week
    );
    if (!currentGroupId || currentGroupId === violation.groupId) {
      continue;
    }

    if (violation.type === "SKILL" && violation.skillId && !worker.skillIds.includes(violation.skillId)) {
      continue;
    }

    const updatedOverrides = setOverrideForWeek(
      baseOverrides,
      worker.id,
      violation.week,
      worker.groupId,
      violation.groupId
    );

    const targetActualAfter =
      violation.type === "HEADCOUNT"
        ? countWorkersWorkingInGroup(
            project,
            assignments,
            violation.groupId,
            violation.week,
            updatedOverrides
          )
        : countWorkersWithSkillWorking(
            project,
            assignments,
            violation.groupId,
            violation.week,
            violation.skillId ?? "",
            updatedOverrides
          );

    if (targetActualAfter <= violation.actual) {
      continue;
    }

    const donorCheck = evaluateGroupWeek(
      project,
      assignments,
      currentGroupId,
      violation.week,
      updatedOverrides
    );

    suggestions.push({
      workerId: worker.id,
      workerName: worker.name,
      fromGroupId: currentGroupId,
      fromGroupName: groupNameById.get(currentGroupId) ?? currentGroupId,
      targetActualAfter,
      remainingMissing: Math.max(0, violation.required - targetActualAfter),
      donorSafe: donorCheck.ok
    });
  }

  return suggestions
    .sort((a, b) => {
      if (a.donorSafe !== b.donorSafe) {
        return a.donorSafe ? -1 : 1;
      }
      if (a.remainingMissing !== b.remainingMissing) {
        return a.remainingMissing - b.remainingMissing;
      }
      return a.workerName.localeCompare(b.workerName);
    })
    .slice(0, 8);
}

      (a, b) => a - b
    );
    return {
      totalRequestWeeks,
      grantedWeeks,
      fulfillmentPercent,
      problemWeeks
    };
  }, [project]);

  useEffect(() => {
    if (!project) {
      return;
    }
    if (!reqGroupId && project.groups.length > 0) {
      setReqGroupId(project.groups[0].id);
    }
  }, [project, reqGroupId]);

  useEffect(() => {
    if (!project) {
      return;
    }
    if (!reqSkillId && project.skills.length > 0) {
      setReqSkillId(project.skills[0].id);
    }
  }, [project, reqSkillId]);

  if (!project) {
    return (
      <div className="space-y-4">
        <Card className="space-y-4">
          <CardTitle>Create project</CardTitle>
          <CardDescription>
            Start with name + year + week range. Data is stored only in local IndexedDB.

              <Label htmlFor="new-name">Project name</Label>

            </div>
            <div className="space-y-1">
              <Label htmlFor="new-year">Year</Label>
              <NumberInput
                id="new-year"
                min={2020}
                max={2100}
                value={newProjectYear}
                onValueChange={setNewProjectYear}
              />
            </div>
            <div className="space-y-1">
              <Label htmlFor="new-start">Start week</Label>
              <NumberInput
                id="new-start"
                min={1}
                max={53}
                value={newProjectStartWeek}
                onValueChange={setNewProjectStartWeek}
              />
            </div>
            <div className="space-y-1">
              <Label htmlFor="new-end">End week</Label>
              <NumberInput
                id="new-end"
                min={1}
                max={53}
                value={newProjectEndWeek}
                onValueChange={setNewProjectEndWeek}
              />
            </div>
          </div>
          <Button
            className="w-full md:w-auto"
            onClick={() =>
              createProject({
                name: newProjectName,
                year: newProjectYear,
                startWeek: newProjectStartWeek,
                endWeek: newProjectEndWeek
              })
            }
          >
            Create project
          </Button>
        </Card>
      </div>
    );
  }

  const selectedWorker = project.workers.find((worker) => worker.id === workerForSkills);

  const renderStep = () => {
    if (wizardStep === 0) {
      return (
        <Card className="space-y-4">
          <CardTitle>Project + time window</CardTitle>
          <CardDescription>Edit project name, year and weeks.</CardDescription>
          <div className="grid gap-3 md:grid-cols-4">
            <div className="space-y-1">
              <Label htmlFor="project-name">Project name</Label>
              <Input
                id="project-name"
                value={project.project.name}
                onChange={(e) => updateProjectName(e.target.value)}
              />
            </div>
            <div className="space-y-1">
              <Label htmlFor="year">Year</Label>
              <NumberInput
                id="year"
                value={project.timeWindow.year}
                onValueChange={(year) => setTimeWindow({ year })}
              />
            </div>
            <div className="space-y-1">
              <Label htmlFor="start">Start week</Label>
              <NumberInput
                id="start"
                min={1}
                max={53}
                value={project.timeWindow.startWeek}
                onValueChange={(startWeek) => setTimeWindow({ startWeek })}
              />
            </div>
            <div className="space-y-1">
              <Label htmlFor="end">End week</Label>
              <NumberInput
                id="end"
                min={1}
                max={53}
                value={project.timeWindow.endWeek}
                onValueChange={(endWeek) => setTimeWindow({ endWeek })}
              />
            </div>
          </div>
          <div className="rounded-xl bg-slate-100 p-3 text-sm">
            <div className="font-semibold">Week count: {weeks.length}</div>
            <div className="mt-1 text-xs text-muted-foreground">Weeks: {weeks.join(", ")}</div>
          </div>
        </Card>
      );
    }

    if (wizardStep === 1) {
      return (

            <Input value={groupName} placeholder="Group name" onChange={(e) => setGroupName(e.target.value)} />
            <NumberInput

    }

    return (
        <Card className="space-y-4">
          <CardTitle>Generate list</CardTitle>
          <CardDescription>
            Deterministic heuristic: priority 1 to 2 to 3, stable by request id.
          </CardDescription>
        <div className="rounded-xl bg-slate-100 p-3 text-xs text-muted-foreground">
          Normal generate uses only WORK/VACATION toggles. Magic generate also attempts automatic
          cross-team backfills for the affected weeks.
        </div>
        <div className="grid gap-2 md:grid-cols-2">
          <div className="rounded-xl bg-slate-100 p-3">
            <div className="text-sm text-muted-foreground">Request fulfillment</div>
            <div className="text-2xl font-bold">{requestSummary.fulfillmentPercent}%</div>
            <div className="text-xs text-muted-foreground">
              granted {requestSummary.grantedWeeks} / requested {requestSummary.totalRequestWeeks} weeks
            </div>
          </div>
          <div className="rounded-xl bg-slate-100 p-3">
            <div className="text-sm text-muted-foreground">Current issues</div>
            <div className="text-2xl font-bold">
              {project.unmetRequests.length} unmet / {project.violations.length} violations
            </div>
            <div className="text-xs text-muted-foreground">
              Problem weeks:{" "}
              {requestSummary.problemWeeks.length ? requestSummary.problemWeeks.join(", ") : "none"}
            </div>
          </div>
        </div>
        <div className="flex flex-wrap gap-2">
          <Button variant="outline" onClick={() => generatePlan()}>
            <Sparkles size={14} className="mr-1" /> Generate lomalista
          </Button>
          <Button onClick={() => void generatePlanWithTransfers()} disabled={isMagicGenerating}>
            <Sparkles size={14} className="mr-1" />
            {isMagicGenerating ? "Generating..." : "Magic generate (with transfers)"}
          </Button>
          <Button variant="outline" onClick={() => setPage("planner")}>
            Go to planner
          </Button>
        </div>
      </Card>
    );
  };

  return (
    <div className="space-y-4">
      <Stepper steps={steps} current={wizardStep} onStepChange={setWizardStep} />
      {renderStep()}
      <div className="flex items-center justify-between">
        <Button variant="outline" onClick={() => setWizardStep(wizardStep - 1)} disabled={wizardStep === 0}>
          <ArrowLeft size={14} className="mr-1" /> Previous
        </Button>
        <Button variant="outline" onClick={() => setWizardStep(wizardStep + 1)} disabled={wizardStep >= 5}>
          Next <ArrowRight size={14} className="ml-1" />
        </Button>
      </div>
    </div>
  );
}

===== D:\Software\ied\Lomalistat\README.md =====

# Lomalistat (Offline-First PWA)

React + TypeScript web app for weekly vacation planning.

## Core guarantees

- Offline-first: runs as a PWA and caches only own build assets.
- Local-only data: all project data is stored in browser IndexedDB.
- No outbound data: no `fetch`, no `axios`, no websocket, no analytics, no telemetry.
- No external assets: no CDN fonts/images/scripts.
- Only transfer path is user-controlled JSON import/export.

## Tech stack

- Vite + React + TypeScript
- Tailwind CSS
- shadcn-style UI components (`src/components/ui`)
- Zustand state store
- IndexedDB via Dexie
- PWA via `vite-plugin-pwa`
- Tests via Vitest

## Run

```bash
npm install
npm run dev
```

Build and preview:

```bash
npm run build
npm run preview
```

Quality checks:

```bash
npm run test
npm run lint
```

## Import / Export

- Import: topbar -> `Import JSON` -> select local `Project.json`.
- Export: topbar -> `Export JSON` -> downloads `<project>_<year>_<start-end>.json`.
- Export PDF: topbar -> `Export PDF` -> choose:
  - vacations-only PDF
  - full report PDF (includes groups, skills, violations)
- Import performs:
  - JSON parse
  - schema migration to `schemaVersion: 1`
  - structure validation
  - save to IndexedDB

## Demo button

Topbar button `Generate demo data (250 ppl)` creates deterministic seeded demo data (`seed=1337`):

- 250 workers
- 4 groups with worker split 50/50/50/100
- 16 skills
- group headcount + skill requirements
- each worker gets exactly one 4-week vacation request
- request window is random within April-September
- demo time window is April-September weeks only (planner columns follow this range)
- deterministic generatePlan run

## Wizard + Planner flow

- Wizard steps:
  1. Project + time window
  2. Groups
  3. Workers
  4. Skills + requirements
  5. Vacation requests
  6. Generate + summary
- Planner:
  - Worker x Week grid (`WORK` / `VACATION`)
  - click cell toggles assignment
  - realtime violation recomputation
  - side panel: headcount, skill coverage, violations, unmet requests

## Storage model

Project JSON is `schemaVersion: 1` and includes:

- project metadata
- time window
- groups, skills, skill requirements
- workers and vacation requests
- plan assignments
- violations and unmet requests

## Directory layout

- `src/model` domain types, validation, migration
- `src/repository` IndexedDB + in-memory repos
- `src/engine` constraints, violations, deterministic generation
- `src/controller` store actions, demo data, import/export
- `src/view` pages/components/layout

## Security/privacy note

The app is intentionally local-only. If browser devtools network tab shows outbound calls, treat it as a bug.

## Dependency security note

- `package.json` uses npm `overrides`:
  - `"serialize-javascript": "^7.0.3"`
- Reason:
  - `workbox-build` (used by `vite-plugin-pwa`) pulls `@rollup/plugin-terser`, which requests `serialize-javascript` `^6.0.1`.
  - npm audit flags `serialize-javascript <=7.0.2` (GHSA-5c6j-r48x-rmvq).
  - Override lifts the resolved version to `7.0.3` while keeping Vite 7 + `vite-plugin-pwa` 1.x compatibility.
